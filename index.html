{ name:'Harvard CURIOSity', pattern:/curiosity\.lib\.harvard\.edu\/[^\/]+\/catalog\/([^\/\?#]+)/, toManifest:async function(url){ var m=url.match(/curiosity\.lib\.harvard\.edu\/[^\/]+\/catalog\/([^\/\?#]+)/); if(!m) return null; var mms=m[1].replace(/^\d+-/,''); // strip Alma network-zone prefix, keep the MMS ID try{ var r=await fetch('https://api.lib.harvard.edu/v2/items.json?q='+encodeURIComponent(mms),{headers:{'Accept':'application/json'}}); if(!r.ok) return null; var text=await r.text(); // DRS URN appears in delivery URLs as urn-3:REPO:NNN; mps wants URN-3 uppercased var urn=text.match(/urn-3:[A-Za-z]+:\d+/i); if(!urn) return null; return 'https://mps.lib.harvard.edu/iiif/3/'+urn[0].replace(/^urn-3/i,'URN-3'); }catch(e){ return null; } } }
